-- ============================================================
-- PREGUNTA A: Categorias/Subcategorias por año y trimestre
-- ============================================================

-- A1. Ventas por Categoria y Año
SELECT
    NON EMPTY {
        [Measures].[Ventas Internet],
        [Measures].[Unidades Vendidas Internet],
        [Measures].[Ganancia Internet]
    } ON COLUMNS,
    NON EMPTY
        [Producto].[Categoría].[Categoría].MEMBERS
        * [Fecha].[Año].[Año].MEMBERS
    ON ROWS
FROM [Ventas Adventure Works];


-- A2. Ventas por Subcategoria por Trimestre (mas detallado)
SELECT
    NON EMPTY {
        [Measures].[Ventas Internet],
        [Measures].[Unidades Vendidas Internet]
    } ON COLUMNS,
    NON EMPTY
        ORDER(
            [Producto].[Categoría - Producto].[Subcategoría].MEMBERS
            * [Fecha].[Trimestre].[Trimestre].MEMBERS,
            [Measures].[Ventas Internet],
            BDESC
        )
    ON ROWS
FROM [Ventas Adventure Works];


-- A3. Tendencia trimestral de la categoria con mas ventas
SELECT
    NON EMPTY [Fecha].[Trimestre].[Trimestre].MEMBERS ON COLUMNS,
    NON EMPTY [Producto].[Categoría].[Categoría].MEMBERS ON ROWS
FROM [Ventas Adventure Works]
WHERE [Measures].[Ventas Internet];


-- ============================================================
-- PREGUNTA B: Rentabilidad por Territorio - Internet vs Reseller
-- ============================================================

-- B1. Comparativa Internet vs Reseller por Pais
SELECT
    NON EMPTY {
        [Measures].[Ventas Internet],
        [Measures].[Ventas Reseller],
        [Measures].[Ganancia Internet],
        [Measures].[Ganancia Reseller]
    } ON COLUMNS,
    NON EMPTY
        ORDER(
            [Territorio de Ventas].[País].[País].MEMBERS,
            [Measures].[Ganancia Internet],
            BDESC
        )
    ON ROWS
FROM [Ventas Adventure Works];


-- B2. Por grupo regional
SELECT
    NON EMPTY {
        [Measures].[Ventas Internet],
        [Measures].[Ventas Reseller],
        [Measures].[Margen Internet Pct],
        [Measures].[Margen Reseller Pct]
    } ON COLUMNS,
    NON EMPTY
        [Territorio de Ventas].[Grupo Regional].[Grupo Regional].MEMBERS
    ON ROWS
FROM [Ventas Adventure Works];


-- B3. Desglose completo: Grupo > Pais > Region
SELECT
    NON EMPTY {
        [Measures].[Ventas Internet],
        [Measures].[Ventas Reseller]
    } ON COLUMNS,
    NON EMPTY
        HIERARCHIZE(
            [Territorio de Ventas].[Geografía].[Grupo Regional].MEMBERS
            + [Territorio de Ventas].[Geografía].[País].MEMBERS
            + [Territorio de Ventas].[Geografía].[Región].MEMBERS
        )
    ON ROWS
FROM [Ventas Adventure Works];


-- ============================================================
-- PREGUNTA C: Top 10 Productos mas rentables por año/trimestre
-- ============================================================

-- C1. Top 10 productos por ganancia bruta (ranking global)
SELECT
    {
        [Measures].[Ventas Internet],
        [Measures].[Costo Producto Internet],
        [Measures].[Ganancia Internet],
        [Measures].[Margen Internet Pct]
    } ON COLUMNS,
    TOPCOUNT(
        [Producto].[Producto].[Producto].MEMBERS,
        10,
        [Measures].[Ganancia Internet]
    ) ON ROWS
FROM [Ventas Adventure Works];


-- C2. Top 10 con distribucion por año
SELECT
    NON EMPTY [Fecha].[Año].[Año].MEMBERS ON COLUMNS,
    TOPCOUNT(
        [Producto].[Producto].[Producto].MEMBERS,
        10,
        [Measures].[Ganancia Internet]
    ) ON ROWS
FROM [Ventas Adventure Works]
WHERE [Measures].[Ganancia Internet];


-- C3. Top 10 con distribucion por año Y trimestre (crossjoin)
SELECT
    {
        [Measures].[Ventas Internet],
        [Measures].[Ganancia Internet]
    } ON COLUMNS,
    NON EMPTY
        TOPCOUNT(
            [Producto].[Producto].[Producto].MEMBERS,
            10,
            [Measures].[Ganancia Internet]
        )
        * [Fecha].[Año].[Año].MEMBERS
        * [Fecha].[Trimestre].[Trimestre].MEMBERS
    ON ROWS
FROM [Ventas Adventure Works];
