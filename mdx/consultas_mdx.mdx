-- ============================================================
-- PREGUNTA A: Categorias/Subcategorias por año y trimestre
-- ============================================================

-- A1. Ventas por Categoria y Año
SELECT 
    NON EMPTY {
        [Measures].[Internet Sales Amount],
        [Measures].[Internet Order Quantity],
        [Measures].[Internet Gross Profit]
    } ON COLUMNS,
    NON EMPTY 
        [Product].[Category].[Category].MEMBERS 
        * [Date].[Calendar Year].[Calendar Year].MEMBERS
    ON ROWS
FROM [Adventure Works];


-- A2. Ventas por Subcategoria por Trimestre (mas detallado)
SELECT 
    NON EMPTY {
        [Measures].[Internet Sales Amount],
        [Measures].[Internet Order Quantity]
    } ON COLUMNS,
    NON EMPTY 
        NONEMPTY(
            [Product].[Product Categories].[Subcategory].MEMBERS
            * [Date].[Calendar Quarter of Year].[Calendar Quarter of Year].MEMBERS
        )
    ON ROWS
FROM [Adventure Works]
ORDER(
    [Product].[Product Categories].[Subcategory].MEMBERS,
    [Measures].[Internet Sales Amount],
    BDESC
);


-- A3. Tendencia trimestral de la categoria con mas ventas
SELECT 
    NON EMPTY [Date].[Calendar Quarter].[Calendar Quarter].MEMBERS ON COLUMNS,
    NON EMPTY [Product].[Category].[Category].MEMBERS ON ROWS
FROM [Adventure Works]
WHERE [Measures].[Internet Sales Amount];


-- ============================================================
-- PREGUNTA B: Rentabilidad por Territorio - Internet vs Reseller
-- ============================================================

-- B1. Comparativa Internet vs Reseller por Pais
SELECT 
    NON EMPTY {
        [Measures].[Internet Sales Amount],
        [Measures].[Reseller Sales Amount],
        [Measures].[Internet Gross Profit],
        [Measures].[Reseller Gross Profit]
    } ON COLUMNS,
    NON EMPTY 
        [Sales Territory].[Sales Territory Country].[Sales Territory Country].MEMBERS
    ON ROWS
FROM [Adventure Works]
ORDER(
    [Sales Territory].[Sales Territory Country].[Sales Territory Country].MEMBERS,
    [Measures].[Internet Gross Profit],
    BDESC
);


-- B2. Por grupo regional
SELECT 
    NON EMPTY {
        [Measures].[Internet Sales Amount],
        [Measures].[Reseller Sales Amount],
        [Measures].[Internet Gross Profit Margin],
        [Measures].[Reseller Gross Profit Margin]
    } ON COLUMNS,
    NON EMPTY 
        [Sales Territory].[Sales Territory Group].[Sales Territory Group].MEMBERS
    ON ROWS
FROM [Adventure Works];


-- B3. Desglose completo: Grupo > Pais > Region
SELECT 
    NON EMPTY {
        [Measures].[Internet Sales Amount],
        [Measures].[Reseller Sales Amount]
    } ON COLUMNS,
    NON EMPTY 
        HIERARCHIZE(
            [Sales Territory].[Sales Territory].[Sales Territory Group].MEMBERS
            + [Sales Territory].[Sales Territory].[Country].MEMBERS
            + [Sales Territory].[Sales Territory].[Region].MEMBERS
        )
    ON ROWS
FROM [Adventure Works];


-- ============================================================
-- PREGUNTA C: Top 10 Productos mas rentables por año/trimestre
-- ============================================================

-- C1. Top 10 productos por ganancia bruta (ranking global)
SELECT 
    NON EMPTY {
        [Measures].[Internet Sales Amount],
        [Measures].[Internet Total Product Cost],
        [Measures].[Internet Gross Profit],
        [Measures].[Internet Gross Profit Margin]
    } ON COLUMNS,
    TOPCOUNT(
        NON EMPTY [Product].[Product].[Product].MEMBERS,
        10,
        [Measures].[Internet Gross Profit]
    ) ON ROWS
FROM [Adventure Works];


-- C2. Top 10 con distribucion por año
SELECT 
    NON EMPTY [Date].[Calendar Year].[Calendar Year].MEMBERS ON COLUMNS,
    TOPCOUNT(
        NON EMPTY [Product].[Product].[Product].MEMBERS,
        10,
        [Measures].[Internet Gross Profit]
    ) ON ROWS
FROM [Adventure Works]
WHERE [Measures].[Internet Gross Profit];


-- C3. Top 10 con distribucion por año Y trimestre (crossjoin)
SELECT 
    NON EMPTY {
        [Measures].[Internet Sales Amount],
        [Measures].[Internet Gross Profit]
    } ON COLUMNS,
    NON EMPTY 
        TOPCOUNT(
            [Product].[Product].[Product].MEMBERS,
            10,
            [Measures].[Internet Gross Profit]
        )
        * [Date].[Calendar Year].[Calendar Year].MEMBERS
        * [Date].[Calendar Quarter of Year].[Calendar Quarter of Year].MEMBERS
    ON ROWS
FROM [Adventure Works]
ORDER(
    [Product].[Product].[Product].MEMBERS,
    [Measures].[Internet Gross Profit],
    BDESC
);
